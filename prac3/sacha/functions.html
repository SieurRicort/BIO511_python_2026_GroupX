def count_above(seq, lim):
    count = 0
    for value in seq:
        if value > lim:
            count += 1
    return count

Summarise a text

def summarize_text(s): # Define the function
# Create dictionary with values = 0. If empty list the loop doesnt work
    summary = {
        'digits' : 0,
        'letters' : 0,
        'others' : 0
    }
# Loop through characters of a string and check their type
    for char in s:
        if char.isdigit():
            summary["digits"] += 1
        elif char.isalpha():
            summary["letters"] += 1
        else:
            summary["others"] += 1
# calculates length of the input
    total = sum(summary.values())
    print(f"{total} characters read through the function while input was {len(s)} characters long.")
    return summary

Exo3: Aggregate with a mode

def aggregate(seq, mode, threshold):
    # condition = parameter mode should be one of those otherwise function stops
    mode_condition = ("sum", "count", "max")
    if mode not in mode_condition:
        raise ValueError("mode parameter should be either sum, count or max.")
    # preset a value for result depending on the input
    if mode == "max":
        result = None 
    else: result = 0
    # core part of the function
    for n in seq:
        if n < 0:
            continue
        if n >= threshold:
            if mode == "sum":
                result += n
            elif mode == "count":
                result += 1
            elif result is None or n > result:
                result = n

    print(f"By using the aggregator at a threshold of {threshold}, the result is {result} for the {mode} mode")
    return result
    

print(result)
