Exo3: while loops vs for loops

# Variables
sequence = 'GATTACAGAACTGATAC' # Create a sample
count_A=0 # How many A did the loop counted so far
position=0 # How many bp did the loop go through
how_many_A = 3 # How many A do i want to count up to

# With while loop
while count_A < how_many_A and position < len(sequence): # Will break once loop found the requested amount of A or when all sample has been analysed
    if sequence[position] == "A":
        count_A += 1
        if count_A == how_many_A:
            print("Found as many As as specified by the user (" + str(how_many_A) + ").")
    position += 1
    if position == len(sequence):
        print ("Reached the end of the sequence (",position,"bp).")
print("In while loop: Found that many As: " + str(count_A))
print("In while loop: Reading sequence went to that bp's position: " + str(position))


# With for
count_A=0
position=0
how_many_A = 10 # How many A do i want to count up to

for bp in sequence:
    if count_A == how_many_A:
        print("Found as many As as specified by the user (" + str(how_many_A) + ").")
        break # Will break loop once it found as many As as requested
    elif bp in ("A"):
        count_A += 1
    position += 1
    if position == len(sequence):
        print ("Reached the end of the sequence (",position,"bp).")
        break # Will break the loop once the whole sample is analysed
print("In for loop: Found that many As: " + str(count_A))
print("In for loop: Reading sequence went to that bp's position: " + str(position))

Exo4: Nested loops

import time
start_time = time.time()

# Variables
sequences = ['ATCTGAGTCCACACATG', 'GCGTCGTGCGATGTTCACGTTGAT', 'CAGTAGTACTCAGT', 'GGTATGCTAGACGAGATCTAATA', 'ATGTTTTTTTTTTAG', 'TAAAGCAGCAGCTGAATGTAA'] # Different sequences
codons = ['CCA', 'TGT', 'GTA', 'TAG'] # Target codons to find

# Script
## Given example
for sequence in sequences:
  for codon in codons:
    if codon in sequence:
      print(codon + " is in " + sequence + " at least once.")

## Nested loops 2.
start_codon = "ATG"
stop_codons = ["TAA", "TAG", "TGA"]

for i, seq in enumerate(sequences): # Enumerates = INT + STR for one item of the list (positition, content)
    start = seq.find(start_codon) # Looking for anything that matches start_codon inside samples from sequences data. Stores it in start
    if start == -1: # This is when there is no sample containing the sequence of start_codon within the data.
        print(f"Sequence {i}: start codon was not found") # Notifies you
        continue # Go through the next iteration without doing downstream checks
    for pos in range(start, len(seq) - 2, 3): # Create a range of "position" for codons the following: start_codon is 3 nt in a row at a given position (start), any other codon is made of 3 nucleotides after that (, 3 at the end) until the 3 last nt of a sample (len(seq) -2).
        codon = seq[pos:pos + 3] # Actually creates the variable that stores the STR of a codon starting from start_codon up to final codon within a sequence in our data
        if codon in stop_codons: # If one of them matches with the list stop_codons -> notifies the user
            print(f"Sequence {i}: start {start_codon} at {start}, "
                  f"stop {codon} at {pos} (ORF length {pos + 3 - start} nt)")
            break # Stop and go for another sequence in our data
    else:  # A start_codon was found but no other codon matched with stop_codons -> notifies the user
        print(f"Sequence {i}: start codon found but no in-frame stop codon")

print("Process finished --- %s seconds ---" % (time.time() - start_time))

Exo6 : Reverse the dictionnary

# /!\ NEEDS name_roots from # Exo5
# 1. Collect unique values in roots_normal through dictionnary
unique_roots = [] # empty list

for roots_list in name_roots.values(): # iterate through values in the dictionary (list)
    for roots_str1 in roots_list: # iterate through items in the list (str)
        if roots_str1 not in unique_roots:
            unique_roots.append(roots_str1) # add a new item (str) in the unique_roots if no other values matches it

print(unique_roots)

# 2. Create the reverse dictionnary
reverse_dict = {} # empty dictionary

# Use the list unique_roots to evaluate each items to the dictionary keys -> add key if unique.
for roots_str2 in unique_roots: # iterate through items in the list (str)
    if roots_str2 not in reverse_dict.keys():
        reverse_dict[roots_str2] = [] # add a new key (str) in reverse_dict if no other values matches it. its value is an empty list. VERY IMPORTANT BECAUSE TO USE APPEND LATER, YOU NEED THE VALUE TO BE OF THE LIST TYPE

# 3. Fill up values with names matching with each unique keys

for name_normal, root_value in name_roots.items(): # go through every items of the dictionnary
    for root_key in root_value: # go through every values (roots) of the first dictionary
            reverse_dict[root_key].append(name_normal) # name_normal is linked to root_normal. reverse dictionary contains all unique roots in the normal dictionary. Therefore, we can append the name of a person (a value) into the corresponding key that is their roots in reverse dictionary.

print(reverse_dict)

# 3.bis: alternative
## reset values of reverse dictionary to not create duplicate values
for root_key in reverse_dict.keys():
    reverse_dict[root_key] = []

for root_key, name_value in reverse_dict.items(): # iterate through dictionnary
    for name_normal, root_value in name_roots.items(): # get both keys and values of name_roots dict
        if root_key in root_value:
            name_value.append(name_normal)

print(reverse_dict)
